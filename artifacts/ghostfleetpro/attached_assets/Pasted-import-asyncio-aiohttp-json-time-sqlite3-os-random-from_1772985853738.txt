# --- 1. DATABASE & STATE ---
DB_FILE = "ghost_pro.db"

def init_db():
    with sqlite3.connect(DB_FILE) as conn:
        c = conn.cursor()
        c.execute("CREATE TABLE IF NOT EXISTS accounts (id TEXT PRIMARY KEY, name TEXT, token TEXT, status TEXT, avatar TEXT)")
        c.execute("CREATE TABLE IF NOT EXISTS rules (id TEXT PRIMARY KEY, label TEXT, keyword TEXT, profile_id TEXT, server_id TEXT, channel_id TEXT, delay TEXT, message TEXT)")
        c.execute("CREATE TABLE IF NOT EXISTS history (id INTEGER PRIMARY KEY, acc_name TEXT, srv_name TEXT, chan_name TEXT, target TEXT, msg TEXT, ts DATETIME DEFAULT CURRENT_TIMESTAMP)")
        c.execute("CREATE TABLE IF NOT EXISTS config (key TEXT PRIMARY KEY, value TEXT)")
        c.execute("CREATE TABLE IF NOT EXISTS cache (profile_id TEXT, server_id TEXT, server_name TEXT, channels TEXT)")
        conn.commit()

init_db()

# --- 2. THE GATEWAY ENGINE ---
async def start_gateway(token, account_id, account_name):
    url = "wss://gateway.discord.gg/?v=10&encoding=json"
    print(f"📡 [GATEWAY] Attempting connection for {account_name}...")
    
    async with aiohttp.ClientSession() as session:
        try:
            async with session.ws_connect(url) as ws:
                # Identify Payload
                await ws.send_json({
                    "op": 2, "d": {
                        "token": token, "capabilities": 16381,
                        "properties": {"os": "Windows", "browser": "Chrome", "device": ""},
                        "presence": {"status": "online", "afk": False},
                        "compress": False, "client_state": {"guild_versions": {}}
                    }
                })

                ACTIVE_SESSIONS[account_id] = True
                
                async for msg in ws:
                    if msg.type == aiohttp.WSMsgType.TEXT:
                        data = json.loads(msg.data)
                        op, t, d = data.get('op'), data.get('t'), data.get('d')

                        if op == 10: # Heartbeat
                            asyncio.create_task(heartbeat(ws, d['heartbeat_interval']))
                        
                        if t == "READY":
                            print(f"✨ [GATEWAY] {account_name} is ONLINE.")

                        if t == "GUILD_CREATE":
                            channels = [{"id": c["id"], "name": c.get("name", "unknown")} for c in d.get("channels", []) if c.get("type") == 0]
                            with sqlite3.connect(DB_FILE) as conn:
                                conn.cursor().execute("INSERT OR REPLACE INTO cache VALUES (?, ?, ?, ?)", (account_id, d["id"], d["name"], json.dumps(channels)))
                                conn.commit()

                        if t == "MESSAGE_CREATE":
                            await evaluate_rules(token, account_id, account_name, d, session)
        except Exception as e:
            print(f"⚠️ [GATEWAY ERROR] {account_name} disconnected: {e}")
        finally:
            ACTIVE_SESSIONS.pop(account_id, None)

async def heartbeat(ws, interval):
    while True:
        await asyncio.sleep(interval / 1000)
        try: 
            await ws.send_json({"op": 1, "d": None})
        except: 
            break

async def evaluate_rules(token, acc_id, acc_name, d, session):
    author = d.get("author", {})
    if not author or author.get("id") == acc_id: return 
    content, cid, gid = d.get("content", "").lower(), d.get("channel_id"), d.get("guild_id")
    
    with sqlite3.connect(DB_FILE) as conn:
        rules = conn.cursor().execute("SELECT * FROM rules WHERE profile_id = ? OR profile_id = 'all'", (acc_id,)).fetchall()
        for r in rules:
            r_id, label, keyword, p_id, s_id, c_id, delay, reply_msg = r
            if keyword and keyword.lower() not in content: continue
            if s_id and s_id != 'all' and str(gid) != s_id: continue
            
            headers = {**BROWSER_HEADERS, "Authorization": token}
            payload = {"content": reply_msg, "message_reference": {"message_id": d["id"]}}
            
            async with session.post(f"https://discord.com/api/v10/channels/{cid}/messages", json=payload, headers=headers) as rep:
                if rep.status == 200:
                    with sqlite3.connect(DB_FILE) as conn:
                        conn.cursor().execute("INSERT INTO history (acc_name, srv_name, chan_name, target, msg) VALUES (?, ?, ?, ?, ?)", (acc_name, str(gid), str(cid), author.get("username"), reply_msg))
                        conn.commit()

# --- 3. API ROUTES ---
@app.post("/api/login")
async def api_login(req: Request):
    try:
        data = await req.json()
        token = data.get("token", "").strip()
        name = data.get("name", "Unnamed Profile")
        
        headers = {**BROWSER_HEADERS, "Authorization": token}

        async with aiohttp.ClientSession() as s:
            async with s.get("https://discord.com/api/v10/users/@me", headers=headers) as r:
                if r.status == 200:
                    user = await r.json()
                    # Save/Update DB
                    with sqlite3.connect(DB_FILE) as conn:
                        conn.cursor().execute("INSERT OR REPLACE INTO accounts VALUES (?, ?, ?, ?, ?)", 
                                             (user["id"], name, token, "Connected", ""))
                        conn.commit()
                    
                    # Spawn Gateway
                    if user["id"] not in ACTIVE_SESSIONS:
                        asyncio.create_task(start_gateway(token, user["id"], name))
                    
                    return {"status": "success", "user": user["username"]}
                else:
                    return JSONResponse({"error": f"Discord Auth Failed ({r.status})"}, status_code=401)
    except Exception as e:
        return JSONResponse({"error": str(e)}, status_code=500)

@app.get("/api/accounts")
async def get_accounts():
    with sqlite3.connect(DB_FILE) as conn:
        conn.row_factory = sqlite3.Row
        return [dict(r) for r in conn.cursor().execute("SELECT id, name, status FROM accounts").fetchall()]

@app.get("/api/history")
async def get_history():
    with sqlite3.connect(DB_FILE) as conn:
        conn.row_factory = sqlite3.Row
        return [dict(r) for r in conn.cursor().execute("SELECT * FROM history ORDER BY ts DESC LIMIT 50").fetchall()]

@app.post("/api/rules")
async def add_rule(req: Request):
    d = await req.json()
    rid = str(int(time.time()))
    with sqlite3.connect(DB_FILE) as conn:
        conn.cursor().execute("INSERT INTO rules VALUES (?,?,?,?,?,?,?,?)", (rid, d['label'], d['keyword'], d['profile'], d['server'], d['channel'], d['delay'], d['message']))
        conn.commit()
    return {"status": "ok"}

@app.get("/api/servers/{acc_id}")
async def get_cached_servers(acc_id: str):
    with sqlite3.connect(DB_FILE) as conn:
        conn.row_factory = sqlite3.Row
        return [dict(r) for r in conn.cursor().execute("SELECT server_id, server_name, channels FROM cache WHERE profile_id = ?", (acc_id,)).fetchall()]

@app.get("/api/stats")
async def get_stats():
    with sqlite3.connect(DB_FILE) as conn:
        ruls = conn.cursor().execute("SELECT COUNT(*) FROM rules").fetchone()[0]
        logs = conn.cursor().execute("SELECT COUNT(*) FROM history").fetchone()[0]
        return {"accounts": len(ACTIVE_SESSIONS), "rules": ruls, "logs": logs}
